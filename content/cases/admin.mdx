```ts
// 예약 게시는 현재 시각 이후만 받는다. 이미 게시된 공지를 고칠 때만 지난 시각을 그대로 둔다.
const noticeSchema = (options: { allowPast: boolean }) =>
  baseSchema.superRefine((value, ctx) => {
    if (!value.scheduled || options.allowPast) return;
    if (isPast(value.publishAt)) {
      ctx.addIssue({ code: 'custom', path: ['publishAt'], message: '현재 시각 이후로 선택해 주세요.' });
    }
  });
```

<Note>실제 코드가 아니라 구조만 단순화한 예시입니다.</Note>
